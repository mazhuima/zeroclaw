3use crate::tools::{ActivatedToolSet, Tool, ToolSpec};
4use anyhow::Result;
5use std::collections::HashSet;
6use std::sync::{Arc, Mutex};
7use zeroclaw_providers::ModelProvider;

9/// Tool specs assembled for one loop iteration.
10pub(crate) struct IterationToolSpecs {
11    pub(crate) tool_specs: Vec<ToolSpec>,
12    pub(crate) known_tool_names: HashSet<String>,
13    pub(crate) use_native_tools: bool,
14}
15
16pub(crate) fn build_iteration_tool_specs(
17    model_provider: &dyn ModelProvider,
18    tools_registry: &[Box<dyn Tool>],
19    excluded_tools: &[String],
20    activated_tools: Option<&Arc<Mutex<ActivatedToolSet>>>,
21) -> Result<IterationToolSpecs> {
22    // Rebuild tool_specs each iteration so newly activated deferred tools appear.
23    let mut tool_specs: Vec<crate::tools::ToolSpec> = tools_registry
24        .iter()
25        .filter(|tool| !excluded_tools.iter().any(|ex| ex == tool.name()))
26        .map(|tool| tool.spec())
27        .collect();
28    if let Some(at) = activated_tools {
29        let activated_tools = match at.lock() {
30            Ok(guard) => guard,
31            Err(poisoned) => {
32                ::zeroclaw_log::record!(
33                    WARN,
34                    ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Note)
35                        .with_outcome(::zeroclaw_log::EventOutcome::Unknown),
36                    "activated-tool lock poisoned while assembling iteration tool specs; recovering guard for read"
37                );
38                poisoned.into_inner()
39            }
40        };
41        for spec in activated_tools.tool_specs() {
42            if !excluded_tools.iter().any(|ex| ex == &spec.name) {
43                tool_specs.push(spec);
44            }
45        }
46    }
47    let known_tool_names: HashSet<String> = tool_specs
48        .iter()
49        .map(|tool| tool.name.to_ascii_lowercase())
50        .collect();
51    let use_native_tools = model_provider.supports_native_tools() && !tool_specs.is_empty();
52
53    Ok(IterationToolSpecs {
54        tool_specs,
55        known_tool_names,
56        use_native_tools,
57    })
58}

159    #[test]
160    fn build_iteration_tool_specs_recovers_poisoned_activated_tool_lock() {
161        let activated = Arc::new(Mutex::new(ActivatedToolSet::new()));
162        let invocations = Arc::new(AtomicUsize::new(0));
163        let activated_tool: Arc<dyn Tool> = Arc::new(CountingTool::new(
164            "docker-mcp__extract_text",
165            Arc::clone(&invocations),
166        ));
167        activated
168            .lock()
169            .unwrap()
170            .activate("docker-mcp__extract_text".into(), activated_tool);
171        let poisoned = Arc::clone(&activated);
172        let _ = std::thread::spawn(move || {
173            let _guard = poisoned.lock().expect("test mutex should lock");
174            panic!("poison activated-tools lock");
175        })
176        .join();
177
178        let specs = build_iteration_tool_specs(&NativeToolsProvider, &[], &[], Some(&activated))
179            .expect("poisoned activated-tools lock should recover for read");
180        assert!(
181            specs
182                .tool_specs
183                .iter()
184                .any(|spec| spec.name == "docker-mcp__extract_text"),
185            "recovered poisoned lock should still expose activated tool specs"
186        );
187    }
