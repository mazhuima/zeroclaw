1//! A [`ModelProvider`] that replays scripted LLM responses from an [`LlmTrace`].
2//!
3//! Promoted from the test-only trace-replay helper so the same deterministic
4//! engine backs both the shipped `zeroclaw eval` command and the test suite.
5
6use async_trait::async_trait;
7use std::collections::VecDeque;
8use std::sync::{Arc, Mutex};
9use zeroclaw_api::attribution::{Attributable, ModelProviderKind, ProviderKind, Role};
10use zeroclaw_api::model_provider::{
11    ChatRequest, ChatResponse, ModelProvider, TokenUsage, ToolCall,
12};
13
14use crate::case::{LlmTrace, TraceResponse};
15
16/// One FIFO queue of scripted steps per conversation turn, plus a cursor marking
17/// the turn currently being replayed.
18struct ReplayState {
19    turns: Vec<VecDeque<TraceResponse>>,
20    current: usize,
21}
22
23/// Replays the steps of an [`LlmTrace`], scoped to one conversation turn at a time.
24///
25/// Each call to [`ModelProvider::chat`] returns the next scripted step **of the
26/// current turn**. Steps are FIFO *within* a turn, but turn boundaries are enforced
27/// rather than flattened: a turn can neither borrow steps from the next one nor
28/// leave its own steps unconsumed.
29///
30/// - `chat` errors if the current turn runs out of steps (the trace *under*-specifies
31///   that turn's LLM round-trips).
32/// - The runner calls [`ReplayHandle::finish_turn`] between turns; it errors if the
33///   finished turn left steps behind (the trace *over*-specifies them).
34///
35/// Either mismatch surfaces as a clear, turn-scoped error instead of silently
36/// bleeding responses across turn boundaries.
37pub struct TraceLlmProvider {
38    state: Arc<Mutex<ReplayState>>,
39    trace_name: String,
40}
41
42impl TraceLlmProvider {
43    /// Build a replay provider from a trace, keeping each turn's steps in its own queue.
44    pub fn from_trace(trace: &LlmTrace) -> Self {
45        let turns = trace
46            .turns
47            .iter()
48            .map(|turn| turn.steps.iter().map(|s| s.response.clone()).collect())
49            .collect();
50        Self {
51            state: Arc::new(Mutex::new(ReplayState { turns, current: 0 })),
52            trace_name: trace.model_name.clone(),
53        }
54    }
55
56    /// A handle the runner uses to advance turn boundaries while it drives the agent.
57    pub fn handle(&self) -> ReplayHandle {
58        ReplayHandle {
59            state: Arc::clone(&self.state),
60            trace_name: self.trace_name.clone(),
61        }
62    }
63}
64
65/// Runner-side handle for advancing the replay cursor between conversation turns.
66///
67/// Shares the provider's queues (the same `Arc` the agent holds), so the runner can
68/// assert per-turn consumption without owning the boxed provider.
69pub struct ReplayHandle {
70    state: Arc<Mutex<ReplayState>>,
71    trace_name: String,
72}
73
74impl ReplayHandle {
75    /// Assert the just-finished turn consumed all of its scripted steps, then advance
76    /// the cursor to the next turn. Errors if any steps were left unconsumed.
77    pub fn finish_turn(&self, turn_index: usize) -> anyhow::Result<()> {
78        let mut state = self.state.lock().unwrap();
79        let leftover = state.turns.get(state.current).map_or(0, |q| q.len());
80        if leftover > 0 {
81            anyhow::bail!(
82                "TraceLlmProvider({}): turn {turn_index} scripted {leftover} step(s) the agent never requested — the trace over-specifies this turn's LLM round-trips",
83                self.trace_name
84            );
85        }
86        state.current += 1;
87        Ok(())
88    }
89}
90
91impl Attributable for TraceLlmProvider {
92    fn role(&self) -> Role {
93        Role::Provider(ProviderKind::Model(ModelProviderKind::Custom))
94    }
95
96    fn alias(&self) -> &str {
97        "eval-replay"
98    }
99}

101#[async_trait]
102impl ModelProvider for TraceLlmProvider {
103    async fn chat_with_system(
104        &self,
105        _system_prompt: Option<&str>,
106        _message: &str,
107        _model: &str,
108        _temperature: Option<f64>,
109    ) -> anyhow::Result<String> {
110        // Not exercised by the agent loop (which uses `chat`); kept for trait completeness.
111        Ok(String::new())
112    }
113
114    async fn chat(
115        &self,
116        _request: ChatRequest<'_>,
117        _model: &str,
118        _temperature: Option<f64>,
119    ) -> anyhow::Result<ChatResponse> {
120        let step = {
121            let mut state = self.state.lock().unwrap();
122            let current = state.current;
123            match state.turns.get_mut(current).and_then(|q| q.pop_front()) {
124                Some(step) => step,
125                None => anyhow::bail!(
126                    "TraceLlmProvider({}): turn {current} requested more LLM responses than the trace provides for that turn",
127                    self.trace_name
128                ),
129            }
130        };
131        match step {
132            TraceResponse::Text {
133                content,
134                input_tokens,
135                output_tokens,
136            } => Ok(ChatResponse {
137                text: Some(content),
138                tool_calls: vec![],
139                usage: Some(TokenUsage {
140                    input_tokens: Some(input_tokens),
141                    output_tokens: Some(output_tokens),
142                    cached_input_tokens: None,
143                }),
144                reasoning_content: None,
145            }),
146            TraceResponse::ToolCalls {
147                tool_calls,
148                input_tokens,
149                output_tokens,
150            } => {
151                let calls = tool_calls
152                    .into_iter()
153                    .map(|tc| ToolCall {
154                        id: tc.id,
155                        name: tc.name,
156                        arguments: serde_json::to_string(&tc.arguments).unwrap_or_default(),
157                        extra_content: None,
158                    })
159                    .collect();
160                Ok(ChatResponse {
161                    text: Some(String::new()),
162                    tool_calls: calls,
163                    usage: Some(TokenUsage {
164                        input_tokens: Some(input_tokens),
165                        output_tokens: Some(output_tokens),
166                        cached_input_tokens: None,
167                    }),
168                    reasoning_content: None,
169                })
170            }
171        }
172    }
173}
